import React, { useEffect, useRef, useState } from "react";
import Peer from "peerjs";
import {
  FaComments,
  FaVideo,
  FaMicrophone,
  FaMicrophoneSlash,
  FaVideoSlash,
} from "react-icons/fa";

const AudioCall = ({
  roomId,
  userId,
  socket,
  setErr,
  setShowChatMD,
  showChatMD,
  users,
  api,
  noneprofile,
}) => {
  const [peerId, setPeerId] = useState("");
  const [peers, setPeers] = useState({});
  const [remoteStreams, setRemoteStreams] = useState({});
  const [isMicOn, setIsMicOn] = useState(true);
  const [isVideoOn, setIsVideoOn] = useState(true);
  const peerInstance = useRef(null);
  const localStreamRef = useRef(null);
  const localVideoRef = useRef(null);
  const retryCount = useRef(0);

  // تابع برای دریافت media (صوت و تصویر)
  const initCall = () => {
    navigator.mediaDevices
      .getUserMedia({ audio: true, video: true })
      .then((stream) => {
        localStreamRef.current = stream;
        // تنظیم ویدئوی لوکال
        if (localVideoRef.current) {
          localVideoRef.current.srcObject = stream;
        }
        socket.emit("ready_for_call", { roomId, userId });
      })
      .catch((error) => {
        console.error("❌ Media Error:", error);
        retryConnection();
        setErr({ message: "ارتباط صوتی/تصویری برقرار نشد !" });
      });
  };

  // تابع تلاش مجدد در صورت خطا
  const retryConnection = () => {
    if (retryCount.current < 3) {
      retryCount.current += 1;
      console.log(
        `🔄 Retrying media connection... Attempt ${retryCount.current}`
      );
      setTimeout(initCall, 2000);
    } else {
      console.error("❌ Failed to establish media connection.");
      setErr({ message: "ارتباط صوتی/تصویری برقرار نشد !" });
    }
  };

  // تابع اتصال به یک peer جدید
  const connectToNewPeer = (remotePeerId, stream) => {
    if (peers[remotePeerId]) return; // اگر قبلاً متصل شده، ادامه نداده شود
    const call = peerInstance.current.call(remotePeerId, stream);
    if (!call) {
      retryConnection();
      setErr({ message: "ارتباط صوتی/تصویری برقرار نشد !" });
      return;
    }
    call.on("stream", (remoteStream) => {
      setRemoteStreams((prev) => ({ ...prev, [remotePeerId]: remoteStream }));
    });
    call.on("close", () => {
      setRemoteStreams((prev) => {
        const updated = { ...prev };
        delete updated[remotePeerId];
        return updated;
      });
      setPeers((prev) => {
        const updated = { ...prev };
        delete updated[remotePeerId];
        return updated;
      });
    });
    setPeers((prev) => ({ ...prev, [remotePeerId]: call }));
  };

  useEffect(() => {
    if (!roomId || !userId || !socket) {
      console.error("❌ Missing roomId, userId, or socket!");
      setErr({ message: "ارتباط صوتی/تصویری برقرار نشد !" });
      return;
    }

    // ایجاد یک instance از PeerJS
    const peer = new Peer(undefined, {
      // تنظیمات اختیاری مانند secure:true در صورت نیاز
    });
    peerInstance.current = peer;

    // دریافت peer id و اطلاع‌رسانی به socket
    peer.on("open", (id) => {
      setPeerId(id);
      socket.emit("peer_id", { roomId, userId, peerId: id });
    });

    // هنگام دریافت تماس از طرف دیگر
    peer.on("call", (call) => {
      if (localStreamRef.current) {
        call.answer(localStreamRef.current);
      }
      call.on("stream", (remoteStream) => {
        setRemoteStreams((prev) => ({ ...prev, [call.peer]: remoteStream }));
      });
      call.on("close", () => {
        setRemoteStreams((prev) => {
          const updated = { ...prev };
          delete updated[call.peer];
          return updated;
        });
        setPeers((prev) => {
          const updated = { ...prev };
          delete updated[call.peer];
          return updated;
        });
      });
      setPeers((prev) => ({ ...prev, [call.peer]: call }));
    });

    // گوش دادن به رویداد socket جهت دریافت peer idهای دیگر
    socket.on("peer_id", ({ senderId, peerId: remotePeerId }) => {
      if (senderId !== userId && localStreamRef.current) {
        connectToNewPeer(remotePeerId, localStreamRef.current);
      }
    });

    // راه‌اندازی media
    initCall();

    return () => {
      socket.off("peer_id");
      if (peerInstance.current) {
        peerInstance.current.disconnect();
        peerInstance.current.destroy();
      }
      if (localStreamRef.current) {
        localStreamRef.current.getTracks().forEach((track) => track.stop());
      }
    };
  }, [roomId, userId, socket, setErr]);

  // تابع تغییر وضعیت میکروفون
  const toggleMic = () => {
    if (localStreamRef.current) {
      localStreamRef.current.getAudioTracks().forEach((track) => {
        track.enabled = !track.enabled;
      });
      setIsMicOn((prev) => !prev);
    }
  };

  // تابع تغییر وضعیت ویدیو
  const toggleVideo = () => {
    if (localStreamRef.current) {
      localStreamRef.current.getVideoTracks().forEach((track) => {
        track.enabled = !track.enabled;
      });
      setIsVideoOn((prev) => !prev);
    }
  };

  // تابع کمکی برای بررسی فعال بودن ویدیو در استریم
  const hasActiveVideo = (stream) => {
    return stream?.getVideoTracks()?.some((track) => track.enabled);
  };

  // دریافت اطلاعات کاربر بر اساس peerId
  const getUserInfo = (peerId) => {
    return users.find((user) => user.peerId === peerId) || {};
  };

  // بررسی آیا حداقل یک ویدیوی فعال وجود دارد
  const hasAnyActiveVideo = Object.values(remoteStreams).some(hasActiveVideo);

  return (
    <div className="relative w-full h-full">
      <div className="flex flex-col">
        {/* ویدئوی محلی */}
        <div className="w-[80%] mx-auto h-[19rem] mb-4 md:mt-[-40px] md:w-[85%] p-2 flex justify-center items-center">
          <video
            ref={localVideoRef}
            autoPlay
            muted
            playsInline
            className="w-[80%] h-auto rounded-2xl bg-gray-800 shadow-md transform transition-transform hover:scale-105"
          />
        </div>

        {/* ویدئوها/پروفایل‌های راه دور */}
        <div className="w-[100%] p-2 flex flex-wrap md:mt-[-30px] justify-center items-center gap-4 md:gap-2">
          {Object.entries(remoteStreams).map(([id, stream]) => {
            const user = getUserInfo(id);
            const isVideoActive = hasActiveVideo(stream);

            return (
              <div
                key={id}
                className="relative w-[45%] md:h-[9rem] flex flex-wrap bg-gray-300 aspect-video rounded-2xl overflow-hidden shadow-lg hover:shadow-xl transition-all"
              >
                {isVideoActive ? (
                  <video
                    autoPlay
                    playsInline
                    ref={(videoEl) => {
                      if (videoEl && stream) videoEl.srcObject = stream;
                    }}
                    className="w-full h-full object-cover"
                  />
                ) : (
                  <div className="w-full h-full flex flex-col items-center justify-center bg-gray-800 p-4">
                    {user.profile_image ? (
                      <img
                        src={user.profile_image}
                        alt={user.name}
                        className="w-20 h-20 md:w-16 md:h-16 rounded-full mb-4 object-cover border-4 border-white"
                      />
                    ) : (
                      <div className="w-20 h-20 md:w-16 md:h-16 rounded-full mb-4 bg-blue-500 flex items-center justify-center text-white text-2xl">
                        {user.name?.[0]?.toUpperCase() || "U"}
                      </div>
                    )}
                    <span className="text-white font-medium text-center">
                      {user.name || "Unknown User"}
                    </span>
                  </div>
                )}

                {/* نشانگر وضعیت ویدیو */}
                <div className="absolute bottom-2 right-2 p-1 bg-black bg-opacity-50 rounded-full">
                  {isVideoActive ? (
                    <FaVideo className="text-green-500 text-sm" />
                  ) : (
                    <FaVideoSlash className="text-red-500 text-sm" />
                  )}
                </div>
              </div>
            );
          })}

          {/* نمایش همه کاربران وقتی هیچ ویدیویی فعال نیست */}
          {!hasAnyActiveVideo &&
            users.map((user) => {
              const isCurrentUser = user.peerId === peerId;
              if (isCurrentUser) return null;

              return (
                <div
                  key={user.peerId}
                  className="relative w-[45%] md:w-[48%] md:h-[9rem] aspect-video bg-zinc-900 rounded-2xl overflow-hidden shadow-lg hover:shadow-xl transition-all"
                >
                  <div className="w-full h-full flex flex-col items-center justify-center p-4">
                    {user.profile_image ? (
                      <img
                        src={`${api}/${user.profile_image}`}
                        alt={user.name}
                        className="w-20 h-20 rounded-full mb-4 md:mb-2 object-cover"
                      />
                    ) : (
                      <img
                        src={`${noneprofile}`}
                        alt={user.name}
                        className="w-20 h-20 rounded-full mb-2 object-cover"
                      />
                    )}
                    <span className="text-white font-medium text-center">
                      {user.name || "Unknown User"}
                    </span>
                    <div className="w-3 h-3 rounded-full bg-gray-300 absolute right-2 top-2"></div>
                  </div>
                </div>
              );
            })}
        </div>
      </div>

      {/* کنترل‌ها */}
      <div className="fixed bottom-0 left-0 right-0 md:w-full md:m-0 w-[53.5%] ml-[24.8%] flex justify-center rounded-t-[90%] gap-6 p-4 bg-black bg-opacity-50">
        {/* کنترل میکروفون */}
        <button
          onClick={toggleMic}
          className="p-3 rounded-full bg-white shadow-md transition-transform transform hover:scale-110 active:scale-95"
          title={isMicOn ? "خاموش کردن میکروفون" : "فعال کردن میکروفون"}
        >
          {isMicOn ? (
            <FaMicrophone className="text-2xl text-gray-800" />
          ) : (
            <FaMicrophoneSlash className="text-2xl text-red-500" />
          )}
        </button>

        {/* کنترل دوربین */}
        <button
          onClick={toggleVideo}
          className="p-3 rounded-full bg-white shadow-md transition-transform transform hover:scale-110 active:scale-95"
          title={isVideoOn ? "خاموش کردن دوربین" : "فعال کردن دوربین"}
        >
          {isVideoOn ? (
            <FaVideo className="text-2xl text-gray-800" />
          ) : (
            <FaVideoSlash className="text-2xl text-red-500" />
          )}
        </button>

        {/* کنترل چت */}
        <button
          onClick={() => setShowChatMD(!showChatMD)}
          className="hidden md:flex p-3 rounded-full bg-blue-500 text-white shadow-lg transition-all duration-300 transform hover:scale-110 active:scale-95 items-center justify-center"
          title="گفتگو"
        >
          <FaComments className="text-2xl" />
        </button>
      </div>
    </div>
  );
};

export default AudioCall;

import React, { useEffect, useRef, useState } from "react";
import Peer from "peerjs";
import { FaComments, FaMicrophone, FaMicrophoneSlash } from "react-icons/fa";
import { motion, AnimatePresence } from "framer-motion";

const AudioCall = ({
  setPeerConnect,
  roomId,
  userId,
  socket,
  setErr,
  setShowChatMD,
  showChatMD,
  users,
  api,
  noneprofile,
}) => {
  const [peerId, setPeerId] = useState("");
  const [peers, setPeers] = useState({});
  const [isMicOn, setIsMicOn] = useState(true);
  const peerInstance = useRef(null);
  const localStreamRef = useRef(null);
  const audioRefs = useRef({});
  const retryCount = useRef(0);

  useEffect(() => {
    if (!roomId || !userId || !socket) {
      console.error("❌ Missing roomId, userId, or socket!");
      setErr({ message: "ارتباط صوتی برقرار نشد !" });
      return;
    }

    const peer = new Peer(undefined, {
      config: {
        iceServers: [
          { urls: "stun:stun.l.google.com:19302" },
        ],
      },
    });

    peerInstance.current = peer;

    peer.on("open", (id) => {
      console.log("🔹 Peer ID:", id);
      setPeerId(id);
      socket.emit("peer_id", { roomId, userId, peerId: id });
    });

    const initCall = async () => {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({
          audio: true,
        });
        localStreamRef.current = stream;
        socket.emit("ready_for_call", { roomId, userId });

        socket.on("peer_id", ({ senderId, peerId }) => {
          if (senderId !== userId) {
            connectToNewPeer(peerId, stream);
          }
        });

        peer.on("call", (call) => {
          call.answer(stream);
          call.on("stream", (remoteStream) => {
            if (!audioRefs.current[call.peer]) {
              const audio = new Audio();
              audio.srcObject = remoteStream;
              audio.autoplay = true;
              audioRefs.current[call.peer] = audio;
              audio.play().catch((error) => {
                console.error("Audio play failed:", error);
                audio.muted = true; // Mute and try again for autoplay issues
                audio.play();
              });
            }
          });
        });

        setPeerConnect(false);
      } catch (error) {
        console.error("❌ Audio Error:", error);
        setErr({ message: "ارتباط صوتی برقرار نشد !" });
        retryConnection();
      }
    };

    const retryConnection = () => {
      if (retryCount.current < 3) {
        retryCount.current += 1;
        console.log(
          `🔄 Retrying audio connection... Attempt ${retryCount.current}`
        );
        setTimeout(initCall, 2000);
      } else {
        console.error("❌ Failed to establish audio connection.");
        setErr({ message: "ارتباط صوتی برقرار نشد !" });
      }
    };

    initCall();

    return () => {
      peer.disconnect();
      peer.destroy();
      if (localStreamRef.current) {
        localStreamRef.current.getTracks().forEach((track) => track.stop());
      }
      Object.values(audioRefs.current).forEach((audio) => {
        audio.pause();
        audio.srcObject = null;
      });
    };
  }, [roomId, userId, socket, setErr]);

  const connectToNewPeer = (remotePeerId, stream) => {
    if (peers[remotePeerId]) return;

    const call = peerInstance.current.call(remotePeerId, stream);
    if (!call) {
      retryConnection();
      setErr({ message: "ارتباط صوتی برقرار نشد !" });
      return;
    }

    call.on("stream", (remoteStream) => {
      if (!audioRefs.current[remotePeerId]) {
        const audio = new Audio();
        audio.srcObject = remoteStream;
        audio.autoplay = true;
        audioRefs.current[remotePeerId] = audio;
        audio.play().catch((error) => {
          console.error("Audio play failed:", error);
          audio.muted = true; // Mute and try again for autoplay issues
          audio.play();
        });
      }
    });

    setPeers((prevPeers) => ({ ...prevPeers, [remotePeerId]: call }));
  };

  const toggleMic = () => {
    if (localStreamRef.current) {
      localStreamRef.current.getAudioTracks().forEach((track) => {
        track.enabled = !track.enabled;
        console.log("Mic status:", track.enabled ? "ON" : "OFF"); // برای دیباگ
      });
      setIsMicOn((prev) => !prev);
    }
  };

  return (
    <div className="relative w-full h-full">
      {/* لودینگ پیشرفته */}

      <div className="flex flex-col h-full">
        {/* لیست کاربران */}
        <div className="flex-1 p-4 container flex-wrap items-center mx-auto w-full gap-4 overflow-auto">
          {users.map((user) => {
            const isCurrentUser = user.peerId === peerId;
            if (isCurrentUser) return null;

            return (
              <motion.div
                key={user.peerId}
                initial={{ opacity: 0, y: 20 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ duration: 0.3 }}
                className="relative users-profile-audio bg-gray-800 rounded-xl p-4 py-6 flex w-[30%] md:w-[47%] flex-col items-center transition-shadow"
              >
                {user.profile_image ? (
                  <img
                    src={`${api}/${user.profile_image}`}
                    alt={user.name}
                    className="w-20 h-20 rounded-full mb-2 object-cover"
                  />
                ) : (
                  <img
                    src={`${noneprofile}`}
                    alt={user.name}
                    className="w-20 h-20 rounded-full mb-2 object-cover"
                  />
                )}
                <span className="text-white font-medium text-center">
                  {user.name || "کاربر ناشناس"}
                </span>
                <div className="absolute top-3 right-2 flex items-center gap-2">
                  <span className="w-3 h-3 rounded-full bg-gray-500 animate-pulse" />
                </div>
              </motion.div>
            );
          })}
        </div>

        {/* کنترل‌ها */}
        <div className="fixed bottom-0 left-0 right-0 md:w-full md:m-0 w-[53.5%] ml-[24.8%] flex justify-center rounded-t-[90%] gap-6 p-4 bg-black bg-opacity-50">
          <motion.button
            whileHover={{ scale: 1.1 }}
            whileTap={{ scale: 0.9 }}
            onClick={toggleMic}
            className={`p-4 rounded-full shadow-lg transition-colors ${
              isMicOn
                ? "bg-green-500 hover:bg-green-600"
                : "bg-red-500 hover:bg-red-600"
            }`}
          >
            {isMicOn ? (
              <FaMicrophone className="text-2xl text-white" />
            ) : (
              <FaMicrophoneSlash className="text-2xl text-white" />
            )}
          </motion.button>

          <motion.button
            whileHover={{ scale: 1.1 }}
            whileTap={{ scale: 0.9 }}
            onClick={() => setShowChatMD(!showChatMD)}
            className="p-4 rounded-full bg-blue-500 hover:bg-blue-600 shadow-lg hidden md:flex"
          >
            <FaComments className="text-2xl text-white" />
          </motion.button>
        </div>
      </div>
    </div>
  );
};

export default AudioCall;
